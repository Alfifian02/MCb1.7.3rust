cmake_minimum_required(VERSION 3.18)
project(mc_cpp CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(CORE_SRC src/core/noise.cpp src/core/terrain.cpp src/core/caves.cpp src/core/mesher.cpp src/core/streamer.cpp src/core/player.cpp)

if(ANDROID)
  # NativeActivity app. Built by android/app/build.gradle (externalNativeBuild) or directly with the NDK toolchain file.
  set(GLUE ${ANDROID_NDK}/sources/android/native_app_glue)
  add_library(native_app_glue STATIC ${GLUE}/android_native_app_glue.c)
  target_include_directories(native_app_glue PUBLIC ${GLUE})
  add_library(mcpp SHARED ${CORE_SRC} src/render/gles_renderer.cpp src/platform/android_main.cpp)
  target_include_directories(mcpp PRIVATE ${GLUE})
  target_link_libraries(mcpp native_app_glue android log EGL GLESv3)
  # Keep the glue's entry point alive through --gc-sections.
  set_target_properties(mcpp PROPERTIES LINK_FLAGS "-u ANativeActivity_onCreate")
else()
  find_package(Threads REQUIRED)
  add_executable(golden_test src/tools/golden_test.cpp ${CORE_SRC})
  add_executable(sim_test src/tools/sim_test.cpp ${CORE_SRC})
  add_executable(softshot src/tools/softshot.cpp ${CORE_SRC})
  foreach(t golden_test sim_test softshot)
    target_link_libraries(${t} Threads::Threads)
  endforeach()
  enable_testing()
  add_test(NAME golden COMMAND golden_test)
  add_test(NAME sim COMMAND sim_test)
endif()
